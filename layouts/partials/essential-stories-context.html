{{- /* essential-stories-context.html Centralised context for "Essential Stories of the Week". Call once per page with the current page context (.) and then read values via: {{ partial "essential-stories-context.html" . }} {{- $es := .Scratch.Get "EssentialStoriesContext" -}} {{- $all := $es.All -}} {{- $recent := $es.Recent -}} {{- $sevenDaysAgo := $es.SevenDaysAgo -}} This avoids repeating the selection and 7‑day window logic across templates. */ -}} {{- if not (.Scratch.Get "EssentialStoriesContext") -}} {{- /* All Essential Stories (no date filter) */ -}} {{- $allPosts := where .Site.RegularPages "Section" "posts" -}} {{- $allEssential := where $allPosts ".Params.categories" "intersect" (slice "Essential Stories of the Week") -}} {{- $allEssential = $allEssential.ByDate.Reverse -}} {{- /* Calendar week (Mon-Sun) = "this week" */ -}} {{- $today := now.Local -}} {{- $weekday := $today.Weekday -}} {{- $daysFromMonday := 0 -}} {{- if eq $weekday 0 -}} {{- /* Sunday is 6 days from Monday */ -}} {{- $daysFromMonday = 6 -}} {{- else -}} {{- /* Monday-Saturday: weekday - 1 */ -}} {{- $daysFromMonday = sub $weekday 1 -}} {{- end -}} {{- $weekStart := $today.AddDate 0 0 (int (mul $daysFromMonday -1)) | time.Format "2006-01-02" | printf "%sT00:00:00Z" | time.AsTime -}} {{- $recent := where $allEssential ".Date" "ge" $weekStart -}} {{- $ctx := dict "All" $allEssential "Recent" $recent "SevenDaysAgo" $weekStart -}} {{- .Scratch.Set "EssentialStoriesContext" $ctx -}} {{- end -}}
